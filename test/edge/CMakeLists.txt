cmake_minimum_required(VERSION 3.20)
project(doubao_edge_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(spdlog CONFIG REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
enable_testing()
set(EDGE_SRC "${CMAKE_CURRENT_LIST_DIR}/../../device/edge/src")
function(edge_test target)
  add_executable(${target} ${ARGN})
  target_include_directories(${target} PRIVATE "${EDGE_SRC}")
  target_link_libraries(${target} PRIVATE spdlog::spdlog Threads::Threads)
endfunction()
edge_test(native_lease_gate native_lease_gate_test.cpp "${EDGE_SRC}/app/native_lease.cpp" "${EDGE_SRC}/wakeup/gate.cpp")
edge_test(wake_listener wake_listener_test.cpp "${EDGE_SRC}/wakeup/local_listener.cpp" "${EDGE_SRC}/wakeup/trigger.cpp" "${EDGE_SRC}/wakeup/gate.cpp")
edge_test(recorder_buffer recorder_buffer_test.cpp "${EDGE_SRC}/audio/recorder_arecord.cpp")
edge_test(event_progress event_progress_test.cpp)
edge_test(memory_store memory_test.cpp)
target_link_libraries(memory_store PRIVATE nlohmann_json::nlohmann_json)
add_test(NAME memory_store COMMAND memory_store)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/fixtures/events" "${CMAKE_BINARY_DIR}/fixtures/recorder")
configure_file(fake_arecord.sh "${CMAKE_BINARY_DIR}/fixtures/recorder/arecord" COPYONLY)
file(CHMOD "${CMAKE_BINARY_DIR}/fixtures/recorder/arecord" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
add_test(NAME native_lease_gate COMMAND sh "${CMAKE_CURRENT_LIST_DIR}/run_native_test.sh" $<TARGET_FILE:native_lease_gate> "${CMAKE_CURRENT_LIST_DIR}/fake_lease.sh")
add_test(NAME wake_listener COMMAND wake_listener)
add_test(NAME recorder_buffer COMMAND recorder_buffer "${CMAKE_BINARY_DIR}/fixtures/recorder")
add_test(NAME event_progress COMMAND event_progress "${CMAKE_BINARY_DIR}/fixtures/events")
add_test(NAME lease_watchdog COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/lease_watch_test.py")
